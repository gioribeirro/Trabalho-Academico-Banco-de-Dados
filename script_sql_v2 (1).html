Visão geral das tabelas Modelo

Nove entidades compõem o banco db_52dp_v2, cobrindo unidade policial, servidores, pessoas envolvidas, ocorrências, inquéritos, depoimentos, evidências e movimentações.

UNIDADE_POLICIAL

  • id_unidade (PK)
  • nome, endereco
  • tipo_unidade, cidade

SERVIDOR

  • id_servidor (PK)
  • id_unidade (FK)
  • nome_ficticio, cargo, matricula_ficticia

PESSOA

  • id_pessoa (PK)
  • nome_ficticio, documento_ficticio
  • observacao, contato_ficticio

NATUREZA_FATO

  • id_natureza (PK)
  • descricao, gravidade

OCORRENCIA

  • id_ocorrencia (PK)
  • id_unidade, id_natureza, id_servidor_registro (FKs)
  • canal_registro, data_registro, localidade, status, relato_fatos

ENVOLVIMENTO_OCORRENCIA

  • id_pessoa + id_ocorrencia (PK composta)
  • papel

INQUERITO

  • id_inquerito (PK)
  • id_ocorrencia (FK)
  • data_abertura, data_conclusao, status, prioridade

DEPOIMENTO

  • id_depoimento (PK)
  • id_inquerito, id_pessoa (FKs)
  • data_depoimento, resumo_ficticio

EVIDENCIA

  • id_evidencia (PK)
  • id_inquerito (FK)
  • tipo_evidencia, descricao, data_apreensao, status_custodia

MOVIMENTACAO

  • id_movimentacao (PK)
  • id_inquerito, id_servidor (FKs)
  • data_movimentacao, tipo_movimentacao, descricao

Criação do banco e tabelas DDL

CREATE DATABASE db_52dp_v2;
USE db_52dp_v2;

CREATE TABLE UNIDADE_POLICIAL (
    id_unidade INT PRIMARY KEY,
    nome VARCHAR(120) NOT NULL,
    endereco VARCHAR(150) NOT NULL,
    tipo_unidade VARCHAR(60) NOT NULL,
    cidade VARCHAR(80) NOT NULL
);

CREATE TABLE SERVIDOR (
    id_servidor INT PRIMARY KEY,
    id_unidade INT NOT NULL,
    nome_ficticio VARCHAR(120) NOT NULL,
    cargo VARCHAR(60) NOT NULL,
    matricula_ficticia VARCHAR(20) NOT NULL,
    CONSTRAINT fk_servidor_unidade FOREIGN KEY (id_unidade)
        REFERENCES UNIDADE_POLICIAL(id_unidade)
);

CREATE TABLE PESSOA (
    id_pessoa INT PRIMARY KEY,
    nome_ficticio VARCHAR(120) NOT NULL,
    documento_ficticio VARCHAR(30) NOT NULL,
    observacao VARCHAR(200),
    contato_ficticio VARCHAR(20) NOT NULL
);

CREATE TABLE NATUREZA_FATO (
    id_natureza INT PRIMARY KEY,
    descricao VARCHAR(100) NOT NULL,
    gravidade VARCHAR(20) NOT NULL
);

CREATE TABLE OCORRENCIA (
    id_ocorrencia INT PRIMARY KEY,
    id_unidade INT NOT NULL,
    id_natureza INT NOT NULL,
    id_servidor_registro INT NOT NULL,
    canal_registro VARCHAR(20) NOT NULL,
    data_registro DATETIME NOT NULL,
    localidade VARCHAR(120) NOT NULL,
    status VARCHAR(30) NOT NULL,
    relato_fatos TEXT NOT NULL,
    CONSTRAINT fk_ocorrencia_unidade FOREIGN KEY (id_unidade)
        REFERENCES UNIDADE_POLICIAL(id_unidade),
    CONSTRAINT fk_ocorrencia_natureza FOREIGN KEY (id_natureza)
        REFERENCES NATUREZA_FATO(id_natureza),
    CONSTRAINT fk_ocorrencia_servidor FOREIGN KEY (id_servidor_registro)
        REFERENCES SERVIDOR(id_servidor)
);

CREATE TABLE ENVOLVIMENTO_OCORRENCIA (
    id_pessoa INT NOT NULL,
    id_ocorrencia INT NOT NULL,
    papel VARCHAR(30) NOT NULL,
    PRIMARY KEY (id_pessoa, id_ocorrencia),
    CONSTRAINT fk_envolvimento_pessoa FOREIGN KEY (id_pessoa)
        REFERENCES PESSOA(id_pessoa),
    CONSTRAINT fk_envolvimento_ocorrencia FOREIGN KEY (id_ocorrencia)
        REFERENCES OCORRENCIA(id_ocorrencia)
);

CREATE TABLE INQUERITO (
    id_inquerito INT PRIMARY KEY,
    id_ocorrencia INT NOT NULL,
    data_abertura DATE NOT NULL,
    data_conclusao DATE,
    status VARCHAR(30) NOT NULL,
    prioridade VARCHAR(20) NOT NULL,
    CONSTRAINT fk_inquerito_ocorrencia FOREIGN KEY (id_ocorrencia)
        REFERENCES OCORRENCIA(id_ocorrencia),
    CONSTRAINT chk_datas_inquerito CHECK (
        data_conclusao IS NULL OR data_conclusao >= data_abertura
    )
);

CREATE TABLE DEPOIMENTO (
    id_depoimento INT PRIMARY KEY,
    id_inquerito INT NOT NULL,
    id_pessoa INT NOT NULL,
    data_depoimento DATE NOT NULL,
    resumo_ficticio VARCHAR(255) NOT NULL,
    CONSTRAINT fk_depoimento_inquerito FOREIGN KEY (id_inquerito)
        REFERENCES INQUERITO(id_inquerito),
    CONSTRAINT fk_depoimento_pessoa FOREIGN KEY (id_pessoa)
        REFERENCES PESSOA(id_pessoa)
);

CREATE TABLE EVIDENCIA (
    id_evidencia INT PRIMARY KEY,
    id_inquerito INT NOT NULL,
    tipo_evidencia VARCHAR(60) NOT NULL,
    descricao VARCHAR(200) NOT NULL,
    data_apreensao DATE NOT NULL,
    status_custodia VARCHAR(30) NOT NULL,
    CONSTRAINT fk_evidencia_inquerito FOREIGN KEY (id_inquerito)
        REFERENCES INQUERITO(id_inquerito)
);

CREATE TABLE MOVIMENTACAO (
    id_movimentacao INT PRIMARY KEY,
    id_inquerito INT NOT NULL,
    id_servidor INT NOT NULL,
    data_movimentacao DATE NOT NULL,
    tipo_movimentacao VARCHAR(60) NOT NULL,
    descricao VARCHAR(200) NOT NULL,
    CONSTRAINT fk_movimentacao_inquerito FOREIGN KEY (id_inquerito)
        REFERENCES INQUERITO(id_inquerito),
    CONSTRAINT fk_movimentacao_servidor FOREIGN KEY (id_servidor)
        REFERENCES SERVIDOR(id_servidor)
);

Consultas SELECT

Consulta simples:

SELECT * FROM OCORRENCIA;

Ocorrência completa (unidade, servidor, natureza, pessoa envolvida):

SELECT o.id_ocorrencia, o.data_registro, o.canal_registro,
    n.descricao AS natureza, u.nome AS unidade_policial,
    s.nome_ficticio AS servidor_registro, p.nome_ficticio AS pessoa,
    eo.papel, o.status
FROM OCORRENCIA o
INNER JOIN NATUREZA_FATO n ON o.id_natureza = n.id_natureza
INNER JOIN UNIDADE_POLICIAL u ON o.id_unidade = u.id_unidade
INNER JOIN SERVIDOR s ON o.id_servidor_registro = s.id_servidor
INNER JOIN ENVOLVIMENTO_OCORRENCIA eo ON o.id_ocorrencia = eo.id_ocorrencia
INNER JOIN PESSOA p ON eo.id_pessoa = p.id_pessoa;

Inquérito e ocorrência de origem:

SELECT i.id_inquerito, i.data_abertura, i.status AS status_inquerito,
    i.prioridade, o.id_ocorrencia, o.canal_registro,
    o.status AS status_ocorrencia, n.descricao AS natureza
FROM INQUERITO i
INNER JOIN OCORRENCIA o ON i.id_ocorrencia = o.id_ocorrencia
INNER JOIN NATUREZA_FATO n ON o.id_natureza = n.id_natureza;

Depoimentos e pessoas:

SELECT d.id_depoimento, d.data_depoimento,
    p.nome_ficticio AS pessoa, d.resumo_ficticio, i.id_inquerito
FROM DEPOIMENTO d
INNER JOIN PESSOA p ON d.id_pessoa = p.id_pessoa
INNER JOIN INQUERITO i ON d.id_inquerito = i.id_inquerito;

Consultas gerenciais Relatórios

Total de ocorrências por natureza:

SELECT n.descricao AS natureza,
    COUNT(o.id_ocorrencia) AS total_ocorrencias
FROM NATUREZA_FATO n
INNER JOIN OCORRENCIA o ON n.id_natureza = o.id_natureza
GROUP BY n.descricao
ORDER BY total_ocorrencias DESC;

Total de ocorrências por canal de registro:

SELECT canal_registro, COUNT(*) AS total_ocorrencias
FROM OCORRENCIA
GROUP BY canal_registro
ORDER BY total_ocorrencias DESC;

Total de inquéritos por status:

SELECT status, COUNT(*) AS total_inqueritos
FROM INQUERITO
GROUP BY status
ORDER BY total_inqueritos DESC;

Auditoria e inquéritos pendentes Controle

Histórico de movimentações (auditoria):

SELECT m.id_movimentacao, m.data_movimentacao, m.tipo_movimentacao,
    m.descricao, i.id_inquerito, s.nome_ficticio AS servidor
FROM MOVIMENTACAO m
INNER JOIN INQUERITO i ON m.id_inquerito = i.id_inquerito
INNER JOIN SERVIDOR s ON m.id_servidor = s.id_servidor
ORDER BY m.data_movimentacao DESC;

Inquéritos sem data de conclusão:

SELECT id_inquerito, id_ocorrencia, data_abertura, status, prioridade
FROM INQUERITO
WHERE data_conclusao IS NULL;
